# BuildWise Group
# Every public page is open to search engines and AI search assistants. Scanner answers
# never appear in a URL a crawler can request: report links keep them after "#", which
# browsers never send to the server, and a page showing a report is marked noindex.
# Drafts under /insights are marked noindex on the page itself until they are approved.
#
# This file is not the whole robots.txt that crawlers receive. Cloudflare's managed
# robots.txt prepends a "Managed Content" block to it in production. That block
# declares the content signals (search permitted, AI training not permitted, use
# restricted to reference) and explicitly disallows the AI training crawlers, such as
# GPTBot, Google-Extended and CCBot. Those crawlers are therefore blocked in production
# even though this file does not name them. Check the live /robots.txt to see the full
# response; Cloudflare, not this repository, controls the managed block.
#
# Search and AI-search crawlers are named explicitly below so the decision is on record.
# A crawler obeys only the most specific group that names it, so each group repeats
# the /api/ rule. OAI-SearchBot, which crawls for ChatGPT Search, is allowed everywhere
# except /api/. ChatGPT-User only fetches a page when a ChatGPT user's request leads
# to it; it is not the automatic crawler for ChatGPT Search.
#
# Robots rules are not the whole story: Cloudflare's bot settings (Block AI bots,
# AI Crawl Control) can also block crawlers before they read this file.

# Google Search (including AI Overviews and AI Mode)
User-agent: Googlebot
Allow: /
Disallow: /api/

# Bing (also used by Copilot and other assistants)
User-agent: Bingbot
Allow: /
Disallow: /api/

# ChatGPT Search crawler: allowed, except /api/
User-agent: OAI-SearchBot
Allow: /
Disallow: /api/

# User-initiated page access from ChatGPT (not the ChatGPT Search crawler)
User-agent: ChatGPT-User
Allow: /
Disallow: /api/

# Perplexity search
User-agent: PerplexityBot
Allow: /
Disallow: /api/

# Claude search
User-agent: Claude-SearchBot
Allow: /
Disallow: /api/

User-agent: *
Allow: /

Disallow: /api/
